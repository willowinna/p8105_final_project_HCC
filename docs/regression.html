Regression Analysis

df = read_csv("data/liver_clean.csv")

Treatment Type

For HCC, how baseline characters influence patients’ treatment intent

df_hcc_model = df |>
  filter(cancer_type == "HCC", !is.na(treatment_intent)) |>
  drop_na(age, sex, ps, tumor_size, tnm_stage, etiology) |>
  mutate(
    curative = ifelse(treatment_intent == "Curative", 1, 0),
    stage_group = ifelse(tnm_stage %in% c("I", "II"), "Early (I-II)", "Advanced (III-IV)"),
    stage_group = factor(stage_group, levels = c("Early (I-II)", "Advanced (III-IV)")),
    sex = factor(sex),
    etiology = factor(etiology) 
  )
fit_hcc_etiology <- glm(
  curative ~ age + sex + ps + tumor_size + stage_group + etiology,
  data = df_hcc_model,
  family = binomial(link = "logit")
)
summary(fit_hcc_etiology)
